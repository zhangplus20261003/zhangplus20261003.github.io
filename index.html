<script>
// ==========在这里修改配置！==========
const repoName = "zhangplus20261003.github.io";
const username = "zhangplus20261003";
const branch = "main";
const scanTargetFolder = "/files"; // 只扫描这个文件夹！
// ===================================

const baseUrl = `https://api.github.com/repos/${username}/${repoName}/contents`;

async function loadFiles(path){
    const res = await fetch(`${baseUrl}${path}`);
    const data = await res.json();
    const container = document.getElementById("fileList");

    const div = document.createElement("div");
    div.className = "folder";
    const title = document.createElement("div");
    title.className = "folder-name";
    title.innerText = path;
    div.appendChild(title);

    data.forEach(item=>{
        const line = document.createElement("div");
        line.className="file-item";
        if(item.type==="dir"){
            line.innerHTML=`📂 ${item.name}`;
            line.onclick=()=>loadFiles(`${path}/${item.name}`);
        }else{
            line.innerHTML=`📄 ${item.name} <a href="${item.download_url}" download>下载</a>`;
        }
        div.appendChild(line);
    })
    container.appendChild(div);
}

// 启动：直接加载指定文件夹
loadFiles(scanTargetFolder);

//搜索功能
document.getElementById("searchInput").oninput=e=>{
    const val = e.target.value.toLowerCase();
    document.querySelectorAll(".file-item").forEach(el=>{
        const text = el.innerText.toLowerCase();
        el.style.display = text.includes(val) ? "flex":"none";
    })
}
</script>
